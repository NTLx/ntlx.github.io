import { getCollection } from 'astro:content';
import ArchiveExplorer from '../../components/ArchiveExplorer.astro';

export const allPosts = (await getCollection(
  'docs',
  ({ id, data }) => id.startsWith('articles/') && Boolean(data.date && data.category),
))
  .sort((a, b) => +b.data.date - +a.data.date);

共 {allPosts.length} 篇文章。可直接按关键词、专题或年份筛选；全站全文搜索仍可使用顶部搜索。

<ArchiveExplorer posts={allPosts} />
