import { getCollection } from 'astro:content';
import ArticleList from '../../components/ArticleList.astro';

export const allPosts = (await getCollection(
  'docs',
  ({ id, data }) => id.startsWith('articles/') && Boolean(data.date && data.category),
))
  .sort((a, b) => +b.data.date - +a.data.date);

export const grouped = (() => {
  const buckets = new Map();
  for (const post of allPosts) {
    const d = post.data.date;
    const key = `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, '0')}`;
    if (!buckets.has(key)) buckets.set(key, []);
    buckets.get(key).push(post);
  }
  return [...buckets.entries()];
})();

共 {allPosts.length} 篇文章。这里是时间线；按主题浏览可从首页或顶部导航进入专题。

{grouped.map(([month, posts]) => (
  <section>
    <h2 id={month}>{month}</h2>
    <ArticleList posts={posts} showDescription={false} />
  </section>
))}
