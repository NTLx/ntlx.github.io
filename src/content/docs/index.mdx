import { LinkCard, CardGrid } from '@astrojs/starlight/components';
import { getCollection } from 'astro:content';
import ArticleList from '../../components/ArticleList.astro';

export const articles = (await getCollection(
  'docs',
  ({ id, data }) => id.startsWith('articles/') && Boolean(data.date && data.category),
))
  .sort((a, b) => +b.data.date - +a.data.date);

export const latestPosts = articles.slice(0, 8);

export const categoryCount = (category) =>
  articles.filter((post) => post.data.category === category).length;
