import { getCollection } from 'astro:content';
import ArticleList from '../../../components/ArticleList.astro';

export const posts = (await getCollection(
  'docs',
  ({ id, data }) => id.startsWith('articles/') && data.category === 'engineering',
))
  .sort((a, b) => +(b.data.date ?? 0) - +(a.data.date ?? 0));

非 AI 的工程实践 — Figma、Stripe、Instacart 等。共 **{posts.length} 篇**，按发布时间倒序排列。

<ArticleList posts={posts} showCategory={false} />
